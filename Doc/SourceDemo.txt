

// ToyCompiler - Bubble Sort Demo
// Input:
//     Array length
//     Random seed
//
// The program generates random integers,
// then sorts them using bubble sort.

class Array
{
    int Length;
    int[] Data;
}

Array Numbers;
int Seed;


/*
    Simple pseudo-random number generator.

    Linear congruential generator:
        Seed = Seed * 1103515245 + 12345

    The generated value is limited to 0~999.
*/
int Random()
{
    Seed = (Seed * 1103515245 + 12345) % 2147483647;

    if(Seed < 0) {Seed = Seed * -1;}
    return Seed % 1000;
}


void FillArray(Array arr)
{
    for(int i = 0; i < arr.Length; i = i + 1)
    {
        arr.Data[i] = Random();
    }
}

void PrintArray(Array arr)
{
    int i = 0;
	while(i < arr.Length)
	{
		Print(arr.Data[i]);
		i=i+1;
	}
}

void BubbleSort(Array arr)
{
    int i;
    int j;
    int temp;

    /*
        Each pass moves the largest remaining
        element to the end of the array.
    */
    for(i = 0; i < arr.Length - 1; i = i + 1)
    {
        for(j = 0; j < arr.Length - 1 - i; j = j + 1)
        {
            if(arr.Data[j] > arr.Data[j + 1])
            {
                temp = arr.Data[j];

                arr.Data[j] = arr.Data[j + 1];

                arr.Data[j + 1] = temp;
            }
        }
    }
}


void Main()
{
    int length;

    Print("Bubble Sort");
    Print("Input array length (1-100):");

    length = Input();

    if(length < 1)
    {
        length = 1;
    }
    else if(length > 100)
    {
        length = 100;
    }

    Print("Input random seed:");

    Seed = Input();

    Numbers = new Array();

    Numbers.Length = length;
    Numbers.Data = new int[length];

    FillArray(Numbers);

    Print("Before sorting:");
    PrintArray(Numbers);

    BubbleSort(Numbers);

    Print("After sorting:");
    PrintArray(Numbers);
}
